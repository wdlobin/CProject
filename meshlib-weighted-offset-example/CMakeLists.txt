cmake_minimum_required(VERSION 3.21)
project(MeshLibWeightedOffsetExample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MESHLIB_ROOT "E:/CLibrary/MeshLibDistVS22_v3.1.4.297" CACHE PATH "MeshLib distribution")
add_executable(WeightedNegativeOffsetTest weighted_negative_offset_test.cpp)
target_include_directories(WeightedNegativeOffsetTest PRIVATE "${MESHLIB_ROOT}/install/include")
target_link_directories(WeightedNegativeOffsetTest PRIVATE "${MESHLIB_ROOT}/install/lib/Release")
target_link_libraries(WeightedNegativeOffsetTest PRIVATE MRVoxels MRMesh)
if(MSVC)
    target_compile_options(WeightedNegativeOffsetTest PRIVATE /bigobj /utf-8)
    target_compile_definitions(WeightedNegativeOffsetTest PRIVATE _ITERATOR_DEBUG_LEVEL=0)
    set_property(TARGET WeightedNegativeOffsetTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()
file(GLOB MESHLIB_DLLS CONFIGURE_DEPENDS "${MESHLIB_ROOT}/install/app/Release/*.dll")
add_custom_command(TARGET WeightedNegativeOffsetTest POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${MESHLIB_DLLS} "$<TARGET_FILE_DIR:WeightedNegativeOffsetTest>"
    COMMAND_EXPAND_LISTS)
enable_testing()
add_test(NAME WeightedNegativeOffset COMMAND WeightedNegativeOffsetTest "${CMAKE_CURRENT_BINARY_DIR}/results")
