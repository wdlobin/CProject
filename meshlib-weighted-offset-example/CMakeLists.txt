cmake_minimum_required(VERSION 3.21)
project(MeshLibWeightedOffsetExample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MESHLIB_ROOT "E:/CLibrary/MeshLibDistVS22_v3.1.4.297" CACHE PATH "MeshLib distribution")
add_executable(WeightedNegativeOffsetTest weighted_negative_offset_test.cpp)
add_executable(PickedFaceShellTest picked_face_shell_test.cpp)

foreach(TARGET_NAME WeightedNegativeOffsetTest PickedFaceShellTest)
    target_include_directories(${TARGET_NAME} PRIVATE "${MESHLIB_ROOT}/install/include")
    target_link_directories(${TARGET_NAME} PRIVATE "${MESHLIB_ROOT}/install/lib/Release")
endforeach()
target_link_libraries(WeightedNegativeOffsetTest PRIVATE MRVoxels MRMesh)
target_link_libraries(PickedFaceShellTest PRIVATE MRMesh)
if(MSVC)
    foreach(TARGET_NAME WeightedNegativeOffsetTest PickedFaceShellTest)
        target_compile_options(${TARGET_NAME} PRIVATE /bigobj /utf-8)
        target_compile_definitions(${TARGET_NAME} PRIVATE _ITERATOR_DEBUG_LEVEL=0)
        set_property(TARGET ${TARGET_NAME} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    endforeach()
endif()
file(GLOB MESHLIB_DLLS CONFIGURE_DEPENDS "${MESHLIB_ROOT}/install/app/Release/*.dll")
foreach(TARGET_NAME WeightedNegativeOffsetTest PickedFaceShellTest)
    add_custom_command(TARGET ${TARGET_NAME} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${MESHLIB_DLLS} "$<TARGET_FILE_DIR:${TARGET_NAME}>"
        COMMAND_EXPAND_LISTS)
endforeach()
enable_testing()
add_test(NAME WeightedNegativeOffset COMMAND WeightedNegativeOffsetTest "${CMAKE_CURRENT_BINARY_DIR}/results")
add_test(NAME PickedFaceShell COMMAND PickedFaceShellTest "${CMAKE_CURRENT_BINARY_DIR}/picked_face_shell_results")
